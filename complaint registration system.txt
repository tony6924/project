#include <stdio.h>
#include <string.h>
#include <stdlib.h>

// Structure to store complaint details
struct Complaint
{
    int sno;
    char name[50];
    char description[200];
};

// Function to read a line of text
void readLine(char str[], int size)
{
    fgets(str, size, stdin);
    str[strcspn(str, "\n")] = '\0';
}

// Function to read an integer
int readInt()
{
    char input[30];
    fgets(input, sizeof(input), stdin);
    return atoi(input);
}

int main()
{
    struct Complaint c[100];
    int count = 0;
    int choice, searchNo, i, found;

    // Display menu repeatedly
    while (1)
    {
        printf("\n--- COMPLAINT REGISTRATION SYSTEM ---\n");
        printf("1. Register Complaint\n");
        printf("2. View All Complaints\n");
        printf("3. Search Complaint\n");
        printf("4. Exit\n");

        printf("Enter your choice: ");
        choice = readInt();

        switch (choice)
        {
            case 1:
                if (count >= 100)
                {
                    printf("Complaint storage is full!\n");
                    break;
                }

                c[count].sno = count + 1;

                printf("Enter your name: ");
                readLine(c[count].name, 50);

                printf("Enter complaint description: ");
                readLine(c[count].description, 200);

                printf("Complaint registered successfully!\n");
                printf("Serial Number: %d\n", c[count].sno);

                count++;
                break;

            case 2:
                if (count == 0)
                {
                    printf("No complaints registered.\n");
                }
                else
                {
                    printf("\n--- ALL COMPLAINTS ---\n");

                    for (i = 0; i < count; i++)
                    {
                        printf("\nSerial Number: %d\n", c[i].sno);
                        printf("Name: %s\n", c[i].name);
                        printf("Complaint: %s\n", c[i].description);
                    }
                }
                break;

            case 3:
                printf("Enter serial number to search: ");
                searchNo = readInt();

                found = 0;

                for (i = 0; i < count; i++)
                {
                    if (c[i].sno == searchNo)
                    {
                        printf("\nComplaint Found!\n");
                        printf("Serial Number: %d\n", c[i].sno);
                        printf("Name: %s\n", c[i].name);
                        printf("Complaint: %s\n", c[i].description);

                        found = 1;
                        break;
                    }
                }

                if (found == 0)
                {
                    printf("Complaint not found.\n");
                }
                break;

            case 4:
                printf("Exiting the program.\n");
                return 0;

            default:
                printf("Invalid choice! Try again.\n");
        }
    }

    return 0;
}
